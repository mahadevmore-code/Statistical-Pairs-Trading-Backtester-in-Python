#z_score define how it work 
current_position = 0
position_list = []

for i in df["z_score"]:
    if current_position == 0 and i > 2:
        current_position = -1
    elif current_position == 0 and i < -2:
        current_position = 1
    elif current_position != 0 and i > -0.5 and i < 0.5:
        current_position = 0
    
    position_list.append(current_position)

df["position"] = position_list
